#
# Purpose: machine-readable transcription of every range-semantics assertion found in
# the Java test suites, for use by a Rust port. Each directive is one assertion.
#
# Transcribed from Apache-2.0 licensed test sources of maven-resolver / maven-artifact:
#   maven-resolver/maven-resolver-util/src/test/java/org/eclipse/aether/util/version/GenericVersionRangeTest.java
#   maven/compat/maven-artifact/src/test/java/org/apache/maven/artifact/versioning/VersionRangeTest.java
# Implementation reference (semantics of the API being described):
#   maven-resolver/maven-resolver-util/src/main/java/org/eclipse/aether/util/version/GenericVersionRange.java
#   maven-resolver/maven-resolver-util/src/main/java/org/eclipse/aether/util/version/VersionSchemeSupport.java
#
# Grammar (one directive per line; '#' starts a comment line; blank lines ignored):
#   contains  <range-spec>  <version>
#   excludes  <range-spec>  <version>
#   invalid   <range-spec>
#   lower     <range-spec>  <version> inclusive|exclusive
#   lower     <range-spec>  none
#   upper     <range-spec>  <version> inclusive|exclusive
#   upper     <range-spec>  none
# Fields are whitespace-separated, EXCEPT on lines that contain a TAB: if a line
# contains at least one TAB character, TAB is the field separator for that line
# (used where a verbatim spec copied from Java contains a space).
#
# NOTE: specs here are *range* specs (GenericVersionRange / a single maven-artifact
# restriction group), not constraint specs. "1.0" is therefore an INVALID range even
# though it is a valid version *constraint* (see constraints.txt).

# === EXTENSION: maven-artifact VersionRange#restrict (intersection) cases ===
# Not expressible in the contains/excludes/invalid/lower/upper vocabulary, so a
# dedicated directive is used. maven-artifact-only semantics: maven-resolver has no
# equivalent operation.
#
#   intersect  <spec1>  <spec2>  <recommended-version|none>  <restrictions>
#
# <spec1>.restrict(<spec2>) yields a range whose recommended version is field 4
# ("none" for null) and whose restriction list is field 5, encoded as the
# comma-joined bracket forms of each restriction ("[lo,hi]", "(lo,hi)", an empty
# side meaning an absent bound, so "(,)" is the unbounded restriction); the literal
# word "empty" means the restriction list is empty.
# Lines whose specs contain a space use TAB as the field separator.
#
# source: VersionRangeTest#testIntersections
